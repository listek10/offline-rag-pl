# offline-rag-pl — zależności DEV (weryfikacja kontraktów)
#
#     python -m venv .venv && source .venv/bin/activate
#     pip install -r requirements-dev.txt         # 61 MB venva, ~5 s bez cache
#     pytest                                      # -> 82 passed, ~0,6 s (stan 29.07)
#
# CO TO UDOWADNIA, A CZEGO NIE — to rozróżnienie jest ważniejsze od samego zielonego koloru:
#   pytest      dowodzi KONTRAKTÓW (kształt odpowiedzi API, kolejność zdarzeń, chunking, metryki,
#               BM25/RRF, routing backendów). Sekundy, bez GPU, bez Qdranta, bez Ollamy.
#   gold set    dowodzi SYSTEMU (src/eval/run_eval.py, eval_answers.py): prawdziwy retrieval
#               i prawdziwy model, pełna ścieżka, liczby. Minuty, wymaga GPU + usług.
# Zielony pytest NIE znaczy "RAG sprawny" — testy podmieniają retrieval i LLM atrapami,
# więc dryf kluczy między atrapą a prawdziwą funkcją złapie dopiero ewaluacja na gold secie.
#
# DLACZEGO ten plik NIE robi `-r requirements.txt`: byłoby to schludniejsze, ale wciągnęłoby
# sentence-transformers, a z nim torcha (~2,5 GB) — a testy torcha nie potrzebują. Nie z ostrożności:
# ciężkie importy (sentence_transformers, qdrant_client, CrossEncoder) siedzą w środku funkcji
# jako leniwe singletony (src/retrieve/search.py, src/index/build_index.py, src/retrieve/rerank.py),
# a `fitz` w parse_pdf.py, którego żaden test nie importuje. Zmierzone: CAŁY zestaw przechodzi
# w venvie zawierającym wyłącznie te cztery paczki poniżej (29.07: 82 testy, 0,5 s).

pytest>=8.0          # runner; testy działają też przez `python tests/test_chunk.py`

# httpx — wymagany przez fastapi.testclient.TestClient (transport in-process, bez gniazda).
# Bez niego tests/test_api.py nie ZBIERA SIĘ (błąd importu), czyli 20 testów przepada w ciszy.
# UWAGA na przyszłość: starlette >=1.3 ostrzega `Using httpx with starlette.testclient is
# deprecated; install httpx2 instead`. Gdy httpx przestanie być wspierany — zamień na `httpx2`.
httpx>=0.27

# fastapi/uvicorn są ŚWIADOMIE zduplikowane z requirements.txt: test_api.py importuje `app`
# i (dla testu streamu) startuje realny serwer w wątku. Duplikacja dwóch lekkich paczek jest
# tańsza niż utrata własności "weryfikacja bez 2,5 GB".
fastapi>=0.115
uvicorn>=0.30
